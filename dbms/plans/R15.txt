Query ID: R15
Type: READ
Description: Tenant list: all active tenants in a property with their room and lease info
Query SQL:
SELECT u.id          AS tenant_id,
       u.name,
       u.email,
       u.phone,
       r.room_number,
       r.floor,
       r.type        AS room_type,
       l.start_date,
       l.end_date,
       l.monthly_rent,
       l.deposit,
       l.is_active
FROM   leases l
JOIN   users u      ON u.id = l.tenant_id
JOIN   rooms r      ON r.id = l.room_id
JOIN   properties p ON p.id = r.property_id
WHERE  p.id = 'property-uuid-here'
  AND  l.is_active = TRUE
ORDER  BY r.room_number;
