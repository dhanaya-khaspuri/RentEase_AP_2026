Query ID: R5
Type: READ
Description: Tenant dashboard: get the active lease for the logged-in tenant
Query SQL:
SELECT l.id           AS lease_id,
       l.start_date,
       l.end_date,
       l.monthly_rent,
       l.deposit,
       l.is_active,
       r.room_number,
       r.floor,
       r.type         AS room_type,
       r.has_ac,
       r.has_attached_bath,
       p.name         AS property_name,
       p.location,
       p.manager_phone
FROM   leases l
JOIN   rooms r      ON r.id = l.room_id
JOIN   properties p ON p.id = r.property_id
WHERE  l.tenant_id = 'tenant-uuid-here'
  AND  l.is_active = TRUE;
