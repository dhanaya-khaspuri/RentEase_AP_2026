Description: Owner payments view: all payments for a specific property
Query SQL:
SELECT pay.id,
       u.name         AS tenant_name,
       u.phone        AS tenant_phone,
       pay.room_number,
       pay.month,
       pay.due_date,
       pay.amount,
       pay.status,
       pay.payment_date,
       pay.payment_method,
       pay.transaction_id
FROM   payments pay
JOIN   users u ON u.id = pay.tenant_id
WHERE  pay.property_id = 'property-uuid-here'
ORDER  BY pay.due_date DESC;
