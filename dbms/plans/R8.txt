Query ID: R8
Type: READ
Description: Overdue payments: all OVERDUE or PENDING-past-due-date payments for a property
Query SQL:
SELECT pay.id,
       u.name         AS tenant_name,
       u.phone,
       pay.room_number,
       pay.month,
       pay.due_date,
       pay.amount,
       pay.status,
       (CURRENT_DATE - pay.due_date) AS days_overdue
FROM   payments pay
JOIN   users u ON u.id = pay.tenant_id
WHERE  pay.property_id = 'property-uuid-here'
  AND  (pay.status = 'OVERDUE'
     OR (pay.status = 'PENDING' AND pay.due_date < CURRENT_DATE))
ORDER  BY days_overdue DESC;
