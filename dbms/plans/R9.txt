Type: READ
Description: Complaints list: all complaints for a property, newest first
Query SQL:
SELECT c.id,
       u.name          AS tenant_name,
       u.phone         AS tenant_phone,
       c.room_number,
       c.category,
       c.title,
       c.description,
       c.priority,
       c.status,
       c.assigned_to,
       c.assigned_phone,
       c.resolution_notes,
       c.created_at,
       c.updated_at
FROM   complaints c
JOIN   users u ON u.id = c.tenant_id
WHERE  c.property_id = 'property-uuid-here'
ORDER  BY
    CASE c.priority
        WHEN 'URGENT' THEN 1
        WHEN 'HIGH'   THEN 2
        WHEN 'MEDIUM' THEN 3
        WHEN 'LOW'    THEN 4
    END,
    c.created_at DESC;
